def graph_laplacian(imputed_train: np.ndarray) ->
    tuple:
    ranks = rankdata(imputed_train[:, :N_CHEM],
    axis=0)          # mid-ranks, ties averaged
    ranks = ranks - ranks.mean(axis=0, keepdims=True)
    norm = np.linalg.norm(ranks, axis=0,
    keepdims=True)
    norm = np.where(norm > 1e-12, norm, 1.0)
    ranks = ranks / norm
    rho = ranks.T @ ranks
    rho = 0.5 * (rho + rho.T)
    non_finite = int((~np.isfinite(rho)).sum())
    rho = np.nan_to_num(rho, nan=0.0, posinf=0.0,
    neginf=0.0)
    np.clip(rho, -1.0, 1.0, out=rho)
    weight = np.where(np.abs(rho) >= TAU, rho, 0.0)
    np.fill_diagonal(weight, 0.0)
    degree = np.abs(weight).sum(axis=1)
    lap231 = np.diag(degree) - weight
    lap231 = 0.5 * (lap231 + lap231.T)
    laplacian = np.zeros((N_CHEM + N_IND, N_CHEM +
    N_IND), dtype=float)
    laplacian[:N_CHEM, :N_CHEM] = lap231
    info = {
        "tau": float(TAU),
        "nodes": int(N_CHEM),
        "edges": int(np.count_nonzero(np.triu(weight,
    1))),
        "isolatedNodes": int(np.count_nonzero(degree
    == 0.0)),
        "minEigenvalue231": float(np.linalg.eigvalsh(
    lap231).min()),
        "nonFiniteRho": non_finite,
        "indicatorBlock": "zero (missing indicators
    are not connected on the chemistry graph)",
    }
    return laplacian, info
def moments(z_train: np.ndarray, y_train: np.ndarray)

class NormalSystem:

    def __init__(self, stats: dict, laplacian:
    np.ndarray, lam: float, gamma: float):
        d = stats["gram"].shape[0]
        self.lam = float(lam)
        self.gamma = float(gamma)
        self.cross = stats["cross"]
        matrix = stats["gram"] + self.lam * np.eye(d)
    + self.gamma * laplacian
        matrix = 0.5 * (matrix + matrix.T)
        self.factor = cho_factor(matrix, lower=True,
    check_finite=False)   # raises if not SPD
        self.solution = cho_solve(self.factor,
    self.cross, check_finite=False)  # A^{-1} C

    def coefficients(self, rank):
        if rank is None:
            return self.solution
        moment = self.cross.T @ self.solution
    # C^T A^{-1} C, q x q PSD
        moment = 0.5 * (moment + moment.T)
        values, vectors = np.linalg.eigh(moment)
        order = np.argsort(values,
    kind="mergesort")[::-1][:int(rank)]
        leading = vectors[:, order]
        return self.solution @ (leading @ leading.T)
    # A^{-1} C Vr Vr^T

    if names != list(FEATURE_COLS):
        raise RuntimeError("q2_chem_meta.json feature
    order differs from target_columns.FEATURE_COLS")
    kind = {record["name"]: record["kind"] for record
    in meta["features"]}
    unknown = sorted(set(kind.values()) -
    {"single_compound", "physical_parameter",
    "derived_sum"})
    if unknown:
        raise RuntimeError(f"unexpected meta kinds:
    {unknown}")

    element = list(C.NON_SINGLE_COMPOUND["element_or_i
    ndex"])
    physical = list(C.NON_SINGLE_COMPOUND["physical_pa
    rameter"])
    derived = list(C.NON_SINGLE_COMPOUND["derived_sum"
    ])
    for group, expected in (("element_or_index",
    element), ("physical_parameter", physical),
                            ("derived_sum", derived)):
        if len(expected) != len(set(expected)):
            raise RuntimeError(f"{group} list has
    duplicates")
        for name in expected:
            if name not in kind:
                raise RuntimeError(f"{group} member
    {name!r} is not a chemistry feature")
            if group == "physical_parameter" and
    kind[name] != "physical_parameter":
                raise RuntimeError(f"{name!r} is
    listed as a physical parameter but meta says
    {kind[name]!r}")
            if group == "derived_sum" and kind[name]
    != "derived_sum":
                raise RuntimeError(f"{name!r} is
    listed as a derived sum but meta says {kind[name]!r}")
            if group == "element_or_index" and
    kind[name] != "single_compound":
                raise RuntimeError(f"{name!r} is an
    element/index feature but meta says {kind[name]!r}")
    reclassified = [name for name in element if
    kind[name] == "single_compound"]

    single = [name for name in FEATURE_COLS if
    kind[name] == "single_compound" and name not in
    element]
    expected_single = 216
    meta_single = sum(1 for name in FEATURE_COLS if
    kind[name] == "single_compound")
    if len(single) != expected_single:
        reason = (f"meta tags {meta_single}
    single_compound features; removing the {len(element)}
    "
                  f"element_or_index members leaves
    {len(single)} (expected {expected_single})")
    else:
        reason = (f"meta tags {meta_single}
    single_compound features; the {len(element)}
    element_or_index "
                  f"members (iron, protein.g.L.,
    betaglucan.mg.L.) are reclassified out of the strict "
                  f"universe, {meta_single} -
    {len(element)} = {len(single)}")
    categories = {"single_compound": single,
    "element_or_index": element,
                  "physical_parameter": physical,
    "derived_sum": derived}
    universe = [name for group in categories.values()
    for name in group]
    if sorted(universe) != sorted(FEATURE_COLS) or
    len(universe) != N_FEATURES:
        raise RuntimeError("the category lists do not
    partition the 231 chemistry features exactly once")
    return {
        "groups": categories,
        "strictNames": single,
        "extendedNames": [name for name in
    FEATURE_COLS if name not in set(single)],
        "kindOf": {name: group for group, members in
    categories.items() for name in members},
        "counts": {group: len(members) for group,
    members in categories.items()},
        "strictSize": len(single),
        "strictSizeExpected": expected_single,
        "strictSizeMatchesExpected": bool(len(single)
    == expected_single),
        "metaSingleCompoundCount": meta_single,
        "metaReclassified": reclassified,
        "reason": reason,
    }
def build_chemistry_mapping(meta: dict, external:
    pd.DataFrame) -> dict:
    features = meta["features"]
    names = [record["name"] for record in features]
    label_map: dict = {}
    for record in features:
        label_map.setdefault(QL.normalise_label(
    record["corrLabel"]), []).append(record["name"])
    name_map = {QL.normalise_label(name): [name] for
    name in names}
    ambiguous_labels = {key: value for key, value in
    label_map.items() if len(value) > 1}

    labels = [str(value) for value in
    external.iloc[:, 0].tolist()]
    normalised = [QL.normalise_label(label) for label
    in labels]
    if len(set(normalised)) != len(normalised):
        raise RuntimeError("6.csv has duplicate
    compound rows after normalisation")

    table, unmatched, ambiguous = [], [], []
    for label, key in zip(labels, normalised):
        candidates = sorted(set(label_map.get(key,
    [])) | set(name_map.get(key, [])))
        if len(candidates) == 1:
            position = len(table)
            name = candidates[0]
            table.append({"row": position, "label":
    label, "feature": name,
                          "kind": None, "corrIndex":
    None, "position": FEATURE_COLS.index(name)})
        elif len(candidates) > 1:
            ambiguous.append({"label": label,
    "candidates": candidates})
        else:
            unmatched.append({"label": label,
    "reason": "no corrLabel and no feature name matches"})
    if unmatched or ambiguous:
        raise RuntimeError(
            f"6.csv chemistry rows are not
    one-to-one: unmatched={unmatched}
    ambiguous={ambiguous}; "
            f"a partial row set would silently change
    the comparison denominator")
    matched_names = [record["feature"] for record in
    table]
    if len(set(matched_names)) != len(matched_names):
        raise RuntimeError("two 6.csv rows map to the
    same chemistry feature")

    kind = {record["name"]: record["kind"] for record
    in features}
    corr = {record["name"]: record["corrIndex"] for
    record in features}
    for record in table:
        record["corrIndex"] = int(
    corr[record["feature"]])
        record["kind"] = kind[record["feature"]]
    if any(record["corrIndex"] != record["position"]
    for record in table):
        raise RuntimeError("meta corrIndex disagrees
    with the FEATURE_COLS position; name and index
    disagree")

    return {
        "source": "6.csv column 1 (display name)
    matched to data/clean/q2_chem_meta.json corrLabel,
    then "
                  "name, after q2_lib.normalise_label
    (lower case, alphanumerics only)",
        "n": len(labels), "matched": len(table),
    "nUnmatched": len(unmatched), "nAmbiguous":
    len(ambiguous),
        "unmatched": unmatched, "ambiguous":
    ambiguous, "oneToOne": True,
        "duplicateNormalisedLabels": [{"label": key,
    "features": value}
                                      for key, value
    in sorted(ambiguous_labels.items())],
        "duplicateNormalisedLabelNote":
            "E-alpha-ionone and E-beta-ionone both
    normalise to 'eionone' (Greek letters are stripped). "
            "No 6.csv row uses that label, so 44/44
    rows resolve; if one ever did, the run aborts instead
    of "
            "picking a feature by position.",
        "table": table,
    }
def build_sensory_mapping(external: pd.DataFrame,
    alignment_path: Path) -> dict:
    alignment = pd.read_csv(alignment_path)
    if list(alignment.columns) != ["external_header",
    "target"]:
        raise RuntimeError(f"{alignment_path} must
    have columns ['external_header', 'target']")
    if len(alignment) != N_DIMENSIONS:
        raise RuntimeError(f"alignment table has
    {len(alignment)} rows, expected {N_DIMENSIONS}")
    align_map = {QL.normalise_label(header):
    str(target)
                 for header, target in
    zip(alignment["external_header"],
    alignment["target"])}
    if len(align_map) != len(alignment):
        raise RuntimeError("alignment table has
    duplicate headers after normalisation")
    if len(set(align_map.values())) != len(align_map):
        raise RuntimeError("alignment table maps two
    headers to one target")
    if set(align_map.values()) != set(C.TARGETS_50):
        raise RuntimeError("alignment table targets
    differ from v6_common.TARGETS_50")
    if align_map != SENSORY_HEADER_TO_TARGET:
        differing = {key: (align_map.get(key),
    SENSORY_HEADER_TO_TARGET.get(key))
                     for key in set(align_map) |
    set(SENSORY_HEADER_TO_TARGET)
                     if align_map.get(key) !=
    SENSORY_HEADER_TO_TARGET.get(key)}
        raise RuntimeError(f"the literal dictionary
    and {alignment_path} disagree: {differing}")

    headers = [str(value) for value in
    external.columns[1:]]
    if len(headers) != N_DIMENSIONS:
        raise RuntimeError(f"6.csv carries
    {len(headers)} sensory columns, expected
    {N_DIMENSIONS}")
    norms = [QL.normalise_label(header) for header in
    headers]
    if len(set(norms)) != len(norms):
        raise RuntimeError("6.csv has duplicate
    sensory headers after normalisation")
    missing = [header for header, key in zip(headers,
    norms) if key not in SENSORY_HEADER_TO_TARGET]
    unused = sorted(set(SENSORY_HEADER_TO_TARGET) -
    set(norms))
    if missing or unused:
        raise RuntimeError(f"6.csv headers do not
    cover the dictionary: missing={missing}
    unused={unused}")
    targets = [SENSORY_HEADER_TO_TARGET[key] for key
    in norms]
    if len(set(targets)) != len(targets) or
    set(targets) != set(C.TARGETS_50):
        raise RuntimeError("6.csv sensory columns are
    not one-to-one onto the 50 targets")

    column_for_target = {target: index for index,
    target in enumerate(targets)}
    positional_mismatch = [{"column": index + 1,
    "header": headers[index],
                            "targetIfPositional":
    C.TARGETS_50[index],
                            "targetIfNamed":
    targets[index]}
                           for index in
    range(len(targets)) if targets[index] !=
    C.TARGETS_50[index]]
    return {
        "source": "explicit literal 50-entry
    dictionary, cross-checked against "
                  "MathorCup_A_独立审查与重跑方案/sensory_name_
    alignment.csv and the 6.csv header set",
        "n": len(headers), "matched": len(headers),
    "oneToOne": True, "unmatched": [], "ambiguous": [],
        "positionalMismatch": positional_mismatch,
        "positionalMismatchCount":
    len(positional_mismatch),
        "columnsForTargets": [column_for_target[target
    ] for target in C.TARGETS_50],
        "table": [{"externalHeader": header,
    "normalised": key, "target": target,
                   "externalColumnIndex": index + 1,
    "targetPosition": C.TARGETS_50.index(target)}
                  for index, (header, key, target) in
    enumerate(zip(headers, norms, targets))],
    }

    enumerate(sorted(names))}

    "rows": rows,
            "nameRank": np.asarray([name_rank[name]
    for name in names], dtype=int),

    float(max(int(value) for value in

def pointscores(frame: pd.DataFrame, tau: float =
    TAU) -> dict:
    imputed = L.imputed_blocks(frame, {"train":
    frame})["train"]
    features = imputed[:, :L.N_FEATURES]
    targets = L.raw_target_matrix(frame)
    signed_total = _guard_matrix(QC.spearman_matrix(
    features, targets), "signed_total")
    basis = QL.style_basis(frame[STYLE_COL].to_numpy(
    ))
    within_signed = _guard_matrix(
        QL.column_correlation(QL.residualise(
    QL.midranks(features), basis),
                              QL.residualise(
    QL.midranks(targets), basis)), "within_signed")
    fitted = L.FittedModel(frame, "zscore",
    GRAPH_SETTINGS["rank"], GRAPH_SETTINGS["lambda"],
                           GRAPH_SETTINGS["gamma"],
    tau=tau)
    beta = _guard_matrix(np.asarray(
    fitted.model["beta"])[:L.N_FEATURES, :], "graph_beta")
    return {"signedTotal": signed_total,
    "total_spearman": np.abs(signed_total),
            "within_signed": within_signed,
    "within_style_spearman": np.abs(within_signed),
            "graph_lowrank": np.abs(beta),
    "graph_beta_signed": beta, "graph": fitted.graph}
def board_ranked(scores: dict, board: dict) -> dict:
    rows = board["rows"]
    ranked = {method: QC.percentile_rank(
    scores[method][rows, :]) for method in BASE_METHODS}
    ranked["consensus"] = (ranked["total_spearman"] +
    ranked["within_style_spearman"]
                           + ranked["graph_lowrank"])
    / 3.0
    for method in METHODS:
        if ranked[method].shape != (board["size"],
    N_DIMENSIONS):
            raise RuntimeError(f"{board['label']}/{met
    hod}: percentile block has the wrong shape")
    return ranked
def order_by_score(ranked: dict, board: dict) -> dict:
    out = {}
    for method in METHODS:
        matrix = ranked[method]
        out[method] = [np.lexsort((board["nameRank"],
    -matrix[:, dimension]))[:K].tolist()
                       for dimension in
    range(N_DIMENSIONS)]
    return out
def order_by_pi(ranked: dict, pi: dict, board: dict)
    -> dict:
    out = {}
    rows = board["rows"]
    for method in METHODS:
        matrix = ranked[method]
        out[method] = [np.lexsort((board["nameRank"],
    -matrix[:, dimension],
                                   -pi[method][rows,
    dimension]))[:K].tolist()
                       for dimension in
    range(N_DIMENSIONS)]
    return out
def sets_as_names(chosen: dict, board: dict) -> dict:
    return {method: [[board["names"][int(local)] for
    local in chosen[method][dimension]]
                     for dimension in
    range(N_DIMENSIONS)] for method in METHODS}
def bootstrap(frame: pd.DataFrame, seed: int, reps:
    int, boards: dict, paths: dict) -> dict:
    styles = frame[STYLE_COL].to_numpy()
    style_order = sorted(set(styles.tolist()))
    groups = [np.nonzero(styles == style)[0] for
    style in style_order]
    if sum(group.size for group in groups) !=
    len(frame):
        raise RuntimeError("style strata do not cover
    the training rows")
    rng = np.random.default_rng(int(seed) + 1000003 *
    21)

    counts = {method: np.zeros((N_FEATURES,
    N_DIMENSIONS), dtype=np.int32) for method in METHODS}
    sign = {key: np.zeros((N_FEATURES, N_DIMENSIONS),
    dtype=np.int32)
            for key in BUCKET_KEYS +
    ("legacyPositive",)}
    handles = {label: open(path, "w",
    encoding="utf-8", newline="") for label, path in
    paths.items()}
    writers = {label: csv.writer(handle,
    lineterminator="\n") for label, handle in
    handles.items()}
    for writer in writers.values():
        writer.writerow(BOOT_HEADER)
    rows_written = {label: 0 for label in paths}
    try:
        for replicate in range(int(reps)):
            indices = np.concatenate(
    [group[rng.integers(0, group.size, size=group.size)]
                                      for group in
    groups])
            sample = frame.iloc[indices].reset_index(
    drop=True)
            scores = pointscores(sample)
            for key, mask in sign_buckets(
    scores["signedTotal"], ZERO_TOL).items():
                sign[key] += mask.astype(np.int32)
            for label, board in boards.items():
                chosen = order_by_score(board_ranked(
    scores, board), board)
                writer = writers[label]
                for method in METHODS:
                    for dimension in
    range(N_DIMENSIONS):
                        for rank, local in
    enumerate(chosen[method][dimension]):
                            counts[method][board["rows
    "][int(local)], dimension] += 1
                            writer.writerow(
    [replicate, int(seed), C.TARGETS_50[dimension],
    method,
                                             rank +
    1, board["names"][int(local)]])
                            rows_written[label] += 1
    finally:
        for handle in handles.values():
            handle.close()
    expected = int(reps) * N_DIMENSIONS *
    len(METHODS) * K
    for label, written in rows_written.items():
        if written != expected:
            raise RuntimeError(f"{label}: wrote
    {written} bootstrap rows, expected {expected}")
    # every cell must be partitioned exactly once per
    replicate; a miss would silently move a denominator
    totals = sum(sign[key] for key in BUCKET_KEYS)
    if not np.array_equal(totals,
    np.full((N_FEATURES, N_DIMENSIONS), int(reps),
    dtype=np.int32)):
        raise RuntimeError("sign buckets do not cover
    every replicate for every (compound, dimension) cell")
    return {"counts": counts, "sign": sign,
    "zeroTol": float(ZERO_TOL), "reps": int(reps),
            "rowsWritten": rows_written,
            "strata": {style: int(group.size) for
    style, group in zip(style_order, groups)} }
def frequencies(counts: dict, reps: int) -> dict:
    return {method: counts[method].astype(float) /
    float(reps) for method in METHODS}
def module_alternatives(method: str, dimension: int,
    module_id: int, board: dict, ranked: dict,
                        delivered: set) -> tuple:
    members = board["byModule"].get(module_id, [])
    if not members:
        return NO_ALTERNATIVE, []
    score = ranked[method]
    ordered = sorted(members, key=lambda name:
    (-float(score[board["index"][name], dimension]),
    name))
    candidates = [name for name in ordered if name
    not in delivered]
    alternative = candidates[0] if candidates else
    NO_ALTERNATIVE
    return alternative, [name for name in
    candidates[:10] if name not in delivered]
def delivered_table(board: dict, ranked: dict, pi:
    dict, sign: dict, scores: dict,
                    modules: dict, kind_of: dict,
    reps: int, zero_tol: float = ZERO_TOL) -> tuple:
    chosen = order_by_pi(ranked, pi, board)
    rows, sets = [], {method: [] for method in
    METHODS}
    for method in METHODS:
        for dimension, target in
    enumerate(C.TARGETS_50):
            picked = [board["names"][int(local)] for
    local in chosen[method][dimension]]
            if len(set(picked)) != K:
                raise RuntimeError(
    f"{board['label']}/{method}/{target}: duplicate names
    in a Top-3 set")
            sets[method].append(picked)
            delivered = set(picked)
            for rank, local in enumerate(
    chosen[method][dimension]):
                name = board["names"][int(local)]
                grow = int(board["rows"][int(local)])
                value = float(pi[method][grow,
    dimension])

    dimension]),
                    int(sign["legacyPositive"][grow,
    dimension]), int(reps),
                    float(scores["signedTotal"][grow,
    dimension]), zero_tol)
                alternative, _pool =
    module_alternatives(
                    method, dimension,
    modules["perCompound"][name]["moduleId"], board,
    ranked, delivered)
                if fields["direction_undetermined"]:
                    status = DIRECTION_UNDETERMINED
                elif value >= PI_STABLE and
    fields["sign_stability"] >= SIGN_STABLE:
                    status = "stable"
                elif value >= PI_STABLE:
                    status = "sign-mixed"
                elif fields["sign_stability"] >=
    SIGN_STABLE:
                    status = "low-pi"
                else:
                    status = "low-pi+sign-mixed"
                rows.append({
                    "dimension": target, "rank": rank
    + 1, "compound": name, "kind": kind_of[name],
                    "method": method,
                    "score": float(ranked[method][int(
    local), dimension]),
                    "pi": value, "pi_mc_se":
    math.sqrt(max(value * (1.0 - value), 0.0) /

                    "sign_stability":
    fields["sign_stability"],
                    # `direction` is kept as a
    TRI-STATE alias of point_direction so the existing
    paper /
                    # figure scripts that read this
    column keep working; the superseded binary rule that
                    # wrote "negative" for a zero or
    missing value is gone.
                    "direction":
    fields["point_direction"],
                    "p_positive":
    fields["p_positive"], "p_negative":
    fields["p_negative"],
                    "p_positive_legacy":
    fields["p_positive_legacy"],
                    "n_positive":
    fields["n_positive"], "n_negative":
    fields["n_negative"],
                    "n_zero": fields["n_zero"],
    "n_invalid": fields["n_invalid"],
                    "point_direction":
    fields["point_direction"],
                    "direction_status":

class GaussianCopula:

    def __init__(self, kappa_R: float, shrink: float
    = 0.05, eig_floor: float = 1e-6):
        self.kappa_R = float(kappa_R)
        self.shrink = float(shrink)
        self.eig_floor = float(eig_floor)
        self.shared = None
        self.factor = {}
        self.corr = {}

    @staticmethod
    def _correlation(Z):
        Z = np.asarray(Z, dtype=float)
        n = Z.shape[0]
        if n < 3:
            return np.eye(Z.shape[1])
        centre = Z - Z.mean(axis=0, keepdims=True)
        scale = centre.std(axis=0)
        scale = np.where(scale > 1e-12, scale, 1.0)
        W = centre / scale
        return (W.T @ W) / float(n - 1)

    def _spd(self, R):
        R = np.asarray(R, dtype=float)
        R = 0.5 * (R + R.T)
        values, vectors = np.linalg.eigh(R)
        values = np.maximum(values, self.eig_floor)
        R = (vectors * values) @ vectors.T
        diag = np.sqrt(np.clip(np.diag(R), 1e-18,
    None))
        R = R / np.outer(diag, diag)
        R = (1.0 - self.shrink) * R + self.shrink *
    np.eye(R.shape[0])
        values, vectors = np.linalg.eigh(R)
        values = np.maximum(values, self.eig_floor)
        R = (vectors * values) @ vectors.T
        diag = np.sqrt(np.clip(np.diag(R), 1e-18,
    None))
        return R / np.outer(diag, diag)

    @staticmethod
    def _cholesky(R):
        try:
            return np.linalg.cholesky(R)
        except np.linalg.LinAlgError:
            for jitter in (1e-10, 1e-8, 1e-6, 1e-4,
    1e-2):
                try:
                    return np.linalg.cholesky(R +
    jitter * np.eye(R.shape[0]))
                except np.linalg.LinAlgError:
                    continue
        raise RuntimeError("copula correlation matrix
    is not positive definite even after jitter")

    def fit(self, Z: np.ndarray, styles) ->
    "GaussianCopula":
        Z = np.asarray(Z, dtype=float)
        styles = np.asarray(styles)
        shared = self._spd(self._correlation(Z))
        self.shared = shared
        self.factor = {}
        for style in sorted(set(str(s) for s in
    styles)):
            rows = np.nonzero(styles == style)[0]
            n_s = rows.size
            eta = n_s / (n_s + self.kappa_R) if n_s +
    self.kappa_R > 0 else 1.0
            empirical = self._correlation(Z[rows]) if
    n_s >= 4 else shared
            R = eta * empirical + (1.0 - eta) * shared
            R = self._spd(R)
            self.corr[str(style)] = R
            self.factor[str(style)] =
    self._cholesky(R)
        return self

    def uniforms_from_normal(self, style: str, z:
    np.ndarray) -> np.ndarray:
        factor = self.factor.get(str(style),
    self._cholesky(self.shared))
        return ndtr(np.asarray(z, dtype=float) @
    factor.T)

    def uniforms(self, style: str, n: int, rng) ->
    np.ndarray:
        z = rng.standard_normal((int(n),
    self.shared.shape[0]))
        return self.uniforms_from_normal(style, z)

def mean_pairwise_distance(A: np.ndarray) -> float:
    A = np.asarray(A, dtype=float)
    n = A.shape[0]
    if n < 2:
        return 0.0
    D = cdist(A, A, metric="euclidean")
    return float(D.sum() / (n * (n - 1)))
def energy_distance_sq(gen: np.ndarray, real:
    np.ndarray, within_fallback: float = None) -> float:
    gen = np.asarray(gen, dtype=float)
    real = np.asarray(real, dtype=float)
    if gen.shape[0] == 0 or real.shape[0] == 0:
        return 0.0
    cross = float(cdist(gen, real,
    metric="euclidean").mean())
    within_gen = mean_pairwise_distance(gen)
    within_real = mean_pairwise_distance(real)
    if within_fallback is not None:
        if gen.shape[0] < 2:
            within_gen = float(within_fallback)
        if real.shape[0] < 2:
            within_real = float(within_fallback)
    return float(2.0 * cross - within_gen -
    within_real)

    metric="euclidean").mean())
